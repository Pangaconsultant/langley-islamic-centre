PART 1: TABLE & COLUMN STRUCTURE SUMMARY
--------------------------------------------------------------------------------

1. public.lic_portal_state (Master Real-Time Portal State & Occupancy Store)
   - id                          text          PRIMARY KEY (Default: 'lic_sl3_main')
   - state                       jsonb         NOT NULL
   - updated_at                  timestamptz   NOT NULL DEFAULT now()

2. public.lic_prayer_timetable (Daily Adhan & Jama'ah Prayer Times)
   - name                        text          PRIMARY KEY ('Fajr', 'Zuhar', 'Asr', 'Maghrib', 'Isha')
   - adhan                       text          NOT NULL
   - jamaat                      text          NOT NULL
   - icon                        text          NOT NULL
   - updated_at                  timestamptz   NOT NULL DEFAULT now()

3. public.lic_visitor_checkins (Indoor Kiosk /kiosk Check-In Log for SBC Audits)
   - id                          text          PRIMARY KEY
   - visitor_name                text          NOT NULL
   - travel_mode                 text          NOT NULL ('Walking', 'Bicycle', 'Car Share', 'Car / Drive')
   - occupant_count              integer       NOT NULL DEFAULT 1
   - activity                    text          NOT NULL ('Fajr', 'Zuhar', 'Asr', 'Maghrib', 'Isha', 'Quran Class', 'Youth Circle', 'Other')
   - checked_in_at               timestamptz   NOT NULL DEFAULT now()

4. public.lic_volunteer_tasks (Flexible Volunteer "Pick-a-Task" Board /volunteer)
   - id                          text          PRIMARY KEY
   - title                       text          NOT NULL
   - category                    text          NOT NULL
   - location                    text          NOT NULL
   - schedule_label              text          NOT NULL
   - duration                    text          NOT NULL
   - status                      text          NOT NULL ('Open' | 'Claimed')
   - claimed_by_name             text          NULL
   - claimed_by_whatsapp         text          NULL

5. public.lic_neighbour_incidents (Neighbour Liaison & Contact Us Enquiries)
   - id                          text          PRIMARY KEY
   - ticket_id                   text          NOT NULL (e.g. 'LIC-NL-2026-XXXX')
   - category                    text          NOT NULL
   - reporter_name               text          NOT NULL
   - status                      text          NOT NULL ('Open' | 'Investigating' | 'Resolved')
   - assigned_lead               text          NOT NULL ('Waqar Ahmed (Operations Lead)')
   - investigation_notes         text          NULL

6. public.lic_quran_students (LIC Maktab Student Registration, Assessment & Attendance)
   - id                          text          PRIMARY KEY
   - reference_no                text          NULL (e.g. 'LIC-MKT-2026-XXXX')
   - session_id                  text          NOT NULL ('session-1' Beginner | 'session-2' Intermediate | 'session-3' Advanced)
   - child_name                  text          NOT NULL
   - date_of_birth               text          NULL
   - age                         integer       NOT NULL
   - school_year_group           text          NULL
   - home_address                text          NULL
   - postcode                    text          NULL
   - medical_or_allergies        text          NULL
   - parent_name                 text          NOT NULL
   - relationship_to_student     text          NULL ('Father' | 'Mother' | 'Guardian' | 'Other')
   - parent_phone                text          NOT NULL
   - parent_email                text          NULL
   - alternative_contact_phone   text          NULL
   - current_quran_level         text          NULL
   - learning_duration           text          NULL
   - attending_other_maktab      boolean       DEFAULT false
   - other_maktab_details        text          NULL
   - previously_assessed_by_lic  boolean       DEFAULT false
   - preferred_days              text          NULL ('Monday – Wednesday' | 'Thursday – Saturday')
   - travel_mode                 text          NOT NULL
   - admission_status            text          DEFAULT 'Confirmed & Enrolled'
   - standing_order_verified     boolean       DEFAULT false
   - digital_signature           text          NULL
   - signature_date              text          NULL
   - attendance                  text          NOT NULL DEFAULT 'Unmarked' ('Unmarked' | 'Present' | 'Late' | 'Absent' | 'Picked Up')
   - enrolled_at                 text          NOT NULL

7. public.lic_staff_directory (Role-Based Approved Staff Allowlist & WebAuthn Passkeys)
   - id                          text          PRIMARY KEY

   - email                       text          NOT NULL UNIQUE

-- Automatic Relational Sync Trigger Function (SECURITY INVOKER)
create or replace function public.sync_lic_portal_state_to_tables()
returns trigger
language plpgsql
security invoker
set search_path = public
as $$
begin
  -- 1. Sync Prayer Timetable
  if jsonb_typeof(new.state->'prayerTimetable') = 'array' then
    insert into public.lic_prayer_timetable (name, adhan, jamaat, icon, updated_at)
    select
      item->>'name',
      item->>'adhan',
      item->>'jamaat',
      coalesce(item->>'icon', 'Sun'),
      now()
    from jsonb_array_elements(new.state->'prayerTimetable') as item
    on conflict (name) do update
      set adhan = excluded.adhan,
          jamaat = excluded.jamaat,
          icon = excluded.icon,
          updated_at = now();
  end if;

  -- 2. Sync Visitor Check-Ins
  if jsonb_typeof(new.state->'checkIns') = 'array' then
    insert into public.lic_visitor_checkins (id, visitor_name, travel_mode, occupant_count, activity, checked_in_at)
    select
      item->>'id',
      item->>'visitorName',
      item->>'travelMode',
      coalesce((item->>'occupantCount')::integer, 1),
      item->>'activity',
      coalesce((item->>'timestamp')::timestamptz, now())
    from jsonb_array_elements(new.state->'checkIns') as item
    on conflict (id) do update
      set visitor_name = excluded.visitor_name,
          travel_mode = excluded.travel_mode,
          occupant_count = excluded.occupant_count,
          activity = excluded.activity;
  end if;

  -- 3. Sync Volunteer Tasks
  if jsonb_typeof(new.state->'volunteerTasks') = 'array' then
    insert into public.lic_volunteer_tasks (id, title, category, location, schedule_label, duration, status, claimed_by_name, claimed_by_whatsapp)
    select
      item->>'id',
      item->>'title',
      item->>'category',
      item->>'location',
      item->>'scheduleLabel',
      item->>'duration',
      item->>'status',
      item->>'claimedByName',
      item->>'claimedByWhatsApp'
    from jsonb_array_elements(new.state->'volunteerTasks') as item
    on conflict (id) do update
      set title = excluded.title,
          category = excluded.category,
          location = excluded.location,
          schedule_label = excluded.schedule_label,
          duration = excluded.duration,
          status = excluded.status,
          claimed_by_name = excluded.claimed_by_name,
          claimed_by_whatsapp = excluded.claimed_by_whatsapp;
  end if;

  -- 4. Sync Neighbour Incidents
  if jsonb_typeof(new.state->'neighbourIncidents') = 'array' then
    insert into public.lic_neighbour_incidents (id, ticket_id, category, reporter_name, status, assigned_lead, investigation_notes)
    select
      item->>'id',
      item->>'ticketId',
      item->>'category',
      item->>'reporterName',
      item->>'status',
      item->>'assignedLead',
      item->>'investigationNotes'
    from jsonb_array_elements(new.state->'neighbourIncidents') as item
    on conflict (id) do update
      set status = excluded.status,
          investigation_notes = excluded.investigation_notes;
  end if;

  -- 5. Sync Quran Students & Maktab Registrations
  if jsonb_typeof(new.state->'quranStudents') = 'array' then
    insert into public.lic_quran_students (
      id, reference_no, session_id, child_name, date_of_birth, age,
      school_year_group, home_address, postcode, medical_or_allergies,
      parent_name, relationship_to_student, parent_phone, parent_email,
      alternative_contact_phone, current_quran_level, learning_duration,
      attending_other_maktab, other_maktab_details, previously_assessed_by_lic,
      preferred_days, travel_mode, admission_status, standing_order_verified,
      digital_signature, signature_date, attendance, enrolled_at
    )
    select
      item->>'id',
      coalesce(item->>'referenceNo', 'LIC-MKT-' || upper(item->>'id')),
      item->>'sessionId',
      item->>'childName',
      item->>'dateOfBirth',
      coalesce((item->>'age')::integer, 8),
      item->>'schoolYearGroup',
      item->>'homeAddress',
      item->>'postcode',
      item->>'medicalOrAllergies',
      item->>'parentName',
      item->>'relationshipToStudent',
      item->>'parentPhone',
      item->>'parentEmail',
      item->>'alternativeContactPhone',
      item->>'currentQuranLevel',
      item->>'learningDuration',
      coalesce((item->>'attendingOtherMaktab')::boolean, false),
      item->>'otherMaktabDetails',
      coalesce((item->>'previouslyAssessed_by_lic')::boolean, false),
      coalesce(item->>'preferredDays', 'Monday – Wednesday'),
      item->>'travelMode',
      coalesce(item->>'admissionStatus', 'Confirmed & Enrolled'),
      coalesce((item->>'standingOrderVerified')::boolean, true),
      item->>'digitalSignature',
      item->>'signatureDate',
      coalesce(item->>'attendance', 'Unmarked'),
      coalesce(item->>'enrolledAt', '2026-09-01')
    from jsonb_array_elements(new.state->'quranStudents') as item
    on conflict (id) do update
      set session_id = excluded.session_id,
          admission_status = excluded.admission_status,
          standing_order_verified = excluded.standing_order_verified,
          attendance = excluded.attendance;
  end if;

  return new;
end;
$$;

create trigger trg_sync_lic_portal_state
after insert or update on public.lic_portal_state
for each row
execute function public.sync_lic_portal_state_to_tables();
